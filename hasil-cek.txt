import { z } from "zod";

export const productQuerySchema = z.object({
  search: z.string().optional(),
  game: z.string().optional(),
  category: z.string().optional(),
  minPrice: z.coerce.number().int().nonnegative().optional(),
  maxPrice: z.coerce.number().int().nonnegative().optional(),
  sort: z.enum(["newest", "oldest", "cheapest", "expensive", "popular", "bestselling"]).default("newest"),
  page: z.coerce.number().int().min(1).default(1),
  pageSize: z.coerce.number().int().min(1).max(48).default(24),
});

export const createProductSchema = z.object({
  name: z.string().min(3),
  gameId: z.string(),
  categoryId: z.string().optional(),
  type: z.enum(["GAME_ACCOUNT", "TOPUP"]),
  description: z.string().optional(),
  specifications: z.record(z.string(), z.unknown()).optional(),
  priceIdr: z.number().int().positive(),
  originalPriceIdr: z.number().int().positive().optional(),
  discountPct: z.number().int().min(0).max(100).optional(),
  stockQty: z.number().int().min(0).default(0),
  warranty: z.string().optional(),
  isFeatured: z.boolean().default(false),
  isBestSeller: z.boolean().default(false),
  isFlashSale: z.boolean().default(false),
  images: z.array(z.string().url()).default([]),
});

export const updateProductSchema = createProductSchema.partial();

import { notFound } from "next/navigation";
import Image from "next/image";
import type { Metadata } from "next";
import { db } from "@/lib/db";
import { formatIdr } from "@/lib/money";
import { AddToCartButton } from "@/components/product/AddToCartButton";
import { ProductJsonLd, BreadcrumbJsonLd } from "@/components/JsonLd";

async function getProduct(slug: string) {
  const product = await db.product.findUnique({
    where: { slug, status: true, deletedAt: null },
    include: {
      images: { orderBy: { position: "asc" } },
      game: true,
      category: true,
      reviews: { where: { isApproved: true }, include: { user: { select: { name: true } } }, take: 20 },
    },
  });
  if (!product) return null;

  const availableStock =
    product.type === "GAME_ACCOUNT"
      ? await db.gameAccount.count({ where: { productId: product.id, status: "AVAILABLE" } })
      : product.stockQty;

  return { ...product, availableStock };
}

export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
  const { slug } = await params;
  const product = await getProduct(slug);
  if (!product) return {};
  return {
    title: product.name,
    description: product.description ?? undefined,
    openGraph: { title: product.name, images: product.images[0] ? [product.images[0].url] : [] },
  };
}

export default async function ProductDetailPage({ params }: { params: Promise<{ slug: string }> }) {
  const { slug } = await params;
  const product = await getProduct(slug);
  if (!product) notFound();

  const specs = (product.specifications as Record<string, string> | null) ?? {};
  const avgRating =
    product.reviews.length > 0
      ? product.reviews.reduce((sum: number, r: any) => sum + r.rating, 0) / product.reviews.length
      : null;

  return (
    <div className="mx-auto max-w-7xl px-4 py-8">
      <ProductJsonLd
        name={product.name}
        description={product.description}
        imageUrl={product.images[0]?.url}
        priceIdr={product.priceIdr}
        inStock={product.availableStock > 0}
      />
      <BreadcrumbJsonLd
        items={[
          { name: "Home", url: "/" },
          { name: product.game.name, url: `/produk?game=${product.game.slug}` },
          { name: product.name, url: `/produk/${product.slug}` },
        ]}
      />
      <div className="grid gap-8 md:grid-cols-2">
        <div>
          <div className="relative aspect-square overflow-hidden rounded-xl2 border border-border bg-background-secondary">
            {product.images[0] ? (
              <Image src={product.images[0].url} alt={product.name} fill className="object-cover" />
            ) : null}
          </div>
          {product.images.length > 1 && (
            <div className="mt-3 flex gap-2 overflow-x-auto scrollbar-none">
              {product.images.map((img: any) => (
                <div key={img.id} className="relative h-16 w-16 shrink-0 overflow-hidden rounded-lg border border-border">
                  <Image src={img.url} alt={product.name} fill className="object-cover" />
                </div>
              ))}
            </div>
          )}
        </div>

        <div>
          <span className="text-xs uppercase tracking-wide text-accent-blue">{product.game.name}</span>
          <h1 className="mt-1 text-xl font-bold md:text-2xl">{product.name}</h1>

          {avgRating && (
            <p className="mt-1 text-sm text-accent-gold">★ {avgRating.toFixed(1)} · {product.reviews.length} ulasan</p>
          )}

          <div className="mt-4 flex items-baseline gap-3">
            <span className="text-2xl font-bold">{formatIdr(product.priceIdr)}</span>
            {product.originalPriceIdr && product.originalPriceIdr > product.priceIdr && (
              <span className="text-sm text-gray-500 line-through">{formatIdr(product.originalPriceIdr)}</span>
            )}
          </div>

          <p className={`mt-2 text-sm font-medium ${product.availableStock > 0 ? "text-emerald-400" : "text-red-400"}`}>
            {product.availableStock > 0 ? `Stok: ${product.availableStock}` : "Stok Habis"}
          </p>

          {Object.keys(specs).length > 0 && (
            <div className="mt-6 rounded-xl2 border border-border bg-background-secondary/60 p-4">
              <h3 className="mb-2 text-sm font-semibold">Spesifikasi</h3>
              <dl className="grid grid-cols-2 gap-y-1.5 text-sm">
                {Object.entries(specs).map(([key, value]) => (
                  <div key={key} className="contents">
                    <dt className="text-gray-400">{key}</dt>
                    <dd className="text-white">{String(value)}</dd>
                  </div>
                ))}
              </dl>
            </div>
          )}

          {product.description && (
            <div className="mt-6">
              <h3 className="mb-2 text-sm font-semibold">Deskripsi</h3>
              <p className="whitespace-pre-line text-sm text-gray-300">{product.description}</p>
            </div>
          )}

          {product.warranty && (
            <p className="mt-4 text-xs text-gray-400">Garansi: {product.warranty}</p>
          )}

          <AddToCartButton
            product={{
              productId: product.id,
              slug: product.slug,
          
quantity: 1,    name: product.name,
              imageUrl: product.images[0]?.url ?? "/placeholder-product.jpg",
              priceIdr: product.priceIdr,
              isUniqueAccount: product.type === "GAME_ACCOUNT",
              maxStock: product.availableStock,
            }}
            disabled={product.availableStock <= 0}
          />
        </div>
      </div>

      {product.reviews.length > 0 && (
        <section className="mt-12">
          <h2 className="mb-4 text-lg font-semibold">Ulasan Pembeli</h2>
          <div className="space-y-3">
            {product.reviews.map((review: any) => (
              <div key={review.id} className="rounded-xl2 border border-border bg-background-secondary/60 p-4">
                <div className="flex items-center justify-between">
                  <span className="text-sm font-medium">{review.user.name ?? "Pengguna"}</span>
                  <span className="text-xs text-accent-gold">★ {review.rating}</span>
                </div>
                {review.comment && <p className="mt-1 text-sm text-gray-300">{review.comment}</p>}
              </div>
            ))}
          </div>
        </section>
      )}
    </div>
  );
}
